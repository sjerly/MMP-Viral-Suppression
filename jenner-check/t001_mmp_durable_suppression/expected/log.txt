154        /*Appying Suppression Rules*/
155        data Long;
156        set labs2;
157        by stateno;
158        retain eversupp losssup firstsupp;
159        if first.stateno then do;
160        	eversupp=0;
161        	losssup=0;
162        	if .<resultn<200 then firstsupp=1; else firstsupp=0;
163        end;
164
165        if resultn lt 200 then eversupp=1;
166        if eversupp=1 and resultn>=200 then losssup=1;
167
168
169        if last.stateno then do;
170        	if eversupp=0 then class=3;				/*People who were never suppressed*/
171        	else if losssup=1 then class=2;			/*People who lost suppression during the time period (unstably suppressed)*/
172        	else if firstsupp=0 then class=1;		/*People who became suppressed during the time period*/
173        	else class=0;							/*People who were suppressed the whole time*/
174        end;
175
176        if class in (0 1) then suppressed_long=1;
177        else suppressed_long=0;
178
179        if last.stateno then output;
180        run;
